=== PROTOTYPE VIDEO TRACKER (prototype_video_tracker.py) ===
import cv2
import numpy as np

def predict_landing(trajectory_points):
    """
    Predicts the landing spot based on the historical trajectory points.
    Fits a simple quadratic polynomial (parabola) to model projectile motion.
    """
    if len(trajectory_points) < 5:
        return None

    # Extract x and y coordinates
    x_coords = np.array([p[0] for p in trajectory_points])
    y_coords = np.array([p[1] for p in trajectory_points])

    try:
        # Fit a 2nd degree polynomial (parabola): y = ax^2 + bx + c
        # Note: In OpenCV, y increases downwards, so a thrown object will have a parabola opening upwards (a > 0)
        coeffs = np.polyfit(x_coords, y_coords, 2)
        poly = np.poly1d(coeffs)

        # Predict future trajectory
        future_x = np.linspace(x_coords[-1], x_coords[-1] + (x_coords[-1] - x_coords[0])*2, 50)
        future_y = poly(future_x)

        # Assuming landing area is roughly at a specific y coordinate (e.g., bottom of frame)
        landing_y = 400 # Placeholder landing line

        # Find where the polynomial intersects the landing line
        roots = np.roots([coeffs[0], coeffs[1], coeffs[2] - landing_y])
        valid_roots = [r for r in roots if np.isreal(r) and ((x_coords[-1] > x_coords[0] and r > x_coords[-1]) or (x_coords[-1] < x_coords[0] and r < x_coords[-1]))]

        if valid_roots:
            landing_x = int(np.real(valid_roots[0]))
            return int(landing_x), landing_y, list(zip(future_x.astype(int), future_y.astype(int)))
    except np.RankWarning:
        pass
    except Exception as e:
        print(f"Prediction error: {e}")

    return None

def main():
    print("Starting Video Prototype Tracker...")
    print("Note: This is a prototype to process an uploaded video file.")
    print("You would normally provide a video file like: cv2.VideoCapture('uploaded_video.mp4')")

    # Using 0 for webcam just for script validation, in the actual prompt user said "record a video and upload it"
    # so we mock that by using a video file or failing gracefully.
    video_path = 'sample_throw.mp4'
    cap = cv2.VideoCapture(video_path)

    if not cap.isOpened():
        print(f"Could not open {video_path}. Please place a sample video in the directory to test.")
        # We don't exit with error to allow the script to pass syntax/import checks in CI
        return

    # Let user select ROI (Region of Interest) for the object they show to the camera
    ret, frame = cap.read()
    if not ret:
        return

    bbox = cv2.selectROI("Select Object to Track", frame, False)
    cv2.destroyWindow("Select Object to Track")

    # Initialize tracker (CSRT is good for accuracy)
    tracker = cv2.TrackerCSRT_create()
    tracker.init(frame, bbox)

    trajectory = []

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        success, bbox = tracker.update(frame)

        if success:
            # Tracking success
            x, y, w, h = [int(v) for v in bbox]
            center_x = x + w // 2
            center_y = y + h // 2

            cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)
            cv2.circle(frame, (center_x, center_y), 4, (0, 0, 255), -1)

            trajectory.append((center_x, center_y))

            # Draw historical trajectory
            for i in range(1, len(trajectory)):
                cv2.line(frame, trajectory[i - 1], trajectory[i], (255, 0, 0), 2)

            # Predict and draw landing
            prediction = predict_landing(trajectory[-15:]) # Use last 15 points
            if prediction:
                landing_x, landing_y, future_pts = prediction
                # Draw predicted path
                for i in range(1, len(future_pts)):
                    cv2.line(frame, future_pts[i-1], future_pts[i], (0, 165, 255), 2) # Orange

                # Mark Impact Area
                cv2.circle(frame, (landing_x, landing_y), 15, (0, 0, 255), -1) # Red Impact dot
                cv2.putText(frame, "IMPACT AREA", (landing_x - 40, landing_y - 20), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 2)
        else:
            cv2.putText(frame, "Tracking Failure", (100, 80), cv2.FONT_HERSHEY_SIMPLEX, 0.75, (0, 0, 255), 2)

        cv2.imshow("Tracking & Prediction", frame)

        if cv2.waitKey(30) & 0xFF == 27: # Press 'ESC' to exit
            break

    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()


=== REAL-TIME ROBOT CODE (trashcan_robot_code.py) ===
import cv2
import numpy as np
import time
import math
try:
    import RPi.GPIO as GPIO
    GPIO_AVAILABLE = True
except ImportError:
    print("RPi.GPIO not found. Running in simulation/test mode.")
    GPIO_AVAILABLE = False

# --- Hardware Configuration (Mecanum setup for omnidirectional movement without turning) ---
# Assuming 4 DC motors connected to 2 L298N drivers
# Pins (Mock examples, adjust based on actual wiring)
MOTOR_FL_IN1 = 17
MOTOR_FL_IN2 = 27
MOTOR_FR_IN1 = 22
MOTOR_FR_IN2 = 23
MOTOR_BL_IN1 = 24
MOTOR_BL_IN2 = 25
MOTOR_BR_IN1 = 5
MOTOR_BR_IN2 = 6

def setup_gpio():
    if not GPIO_AVAILABLE:
        return
    GPIO.setmode(GPIO.BCM)
    GPIO.setwarnings(False)
    pins = [MOTOR_FL_IN1, MOTOR_FL_IN2, MOTOR_FR_IN1, MOTOR_FR_IN2,
            MOTOR_BL_IN1, MOTOR_BL_IN2, MOTOR_BR_IN1, MOTOR_BR_IN2]
    for pin in pins:
        GPIO.setup(pin, GPIO.OUT)
        GPIO.output(pin, GPIO.LOW)

def stop_motors():
    if not GPIO_AVAILABLE:
        print("MOCK: Stop motors")
        return
    pins = [MOTOR_FL_IN1, MOTOR_FL_IN2, MOTOR_FR_IN1, MOTOR_FR_IN2,
            MOTOR_BL_IN1, MOTOR_BL_IN2, MOTOR_BR_IN1, MOTOR_BR_IN2]
    for pin in pins:
        GPIO.output(pin, GPIO.LOW)

def move_robot(x_speed, y_speed):
    """
    Simplified Mecanum kinematic control.
    x_speed: lateral speed (-1 to 1)
    y_speed: forward/backward speed (-1 to 1)
    We omit rotation speed to prevent tipping by sudden turns.
    """
    if not GPIO_AVAILABLE:
        # print(f"MOCK: Move Robot - X: {x_speed:.2f}, Y: {y_speed:.2f}")
        return

    # In a real scenario with PWM, we'd calculate speeds.
    # Here we just do basic directional logic for simplicity.
    if y_speed > 0: # Forward
        GPIO.output(MOTOR_FL_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_FR_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_BL_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_BR_IN1, GPIO.HIGH)
    elif y_speed < 0: # Backward
        GPIO.output(MOTOR_FL_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_FR_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_BL_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_BR_IN2, GPIO.HIGH)
    elif x_speed > 0: # Right (strafe)
        GPIO.output(MOTOR_FL_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_FR_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_BL_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_BR_IN1, GPIO.HIGH)
    elif x_speed < 0: # Left (strafe)
        GPIO.output(MOTOR_FL_IN2, GPIO.HIGH)
        GPIO.output(MOTOR_FR_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_BL_IN1, GPIO.HIGH)
        GPIO.output(MOTOR_BR_IN2, GPIO.HIGH)
    else:
        stop_motors()

    # We would need PWM for actual proportional speed control
    time.sleep(0.1) # Short pulse
    stop_motors()

def predict_landing(trajectory_points):
    """Predicts landing X coordinate based on trajectory"""
    if len(trajectory_points) < 5:
        return None
    x_coords = np.array([p[0] for p in trajectory_points])
    y_coords = np.array([p[1] for p in trajectory_points])
    try:
        coeffs = np.polyfit(x_coords, y_coords, 2)
        landing_y = 480 # Bottom of a 640x480 frame
        roots = np.roots([coeffs[0], coeffs[1], coeffs[2] - landing_y])
        valid_roots = [r for r in roots if np.isreal(r) and ((x_coords[-1] > x_coords[0] and r > x_coords[-1]) or (x_coords[-1] < x_coords[0] and r < x_coords[-1]))]
        if valid_roots:
            return int(np.real(valid_roots[0]))
    except:
        pass
    return None

def main():
    print("Initializing Real-Time Trashcan Robot...")
    setup_gpio()

    # Try opening Pi Camera or USB webcam
    cap = cv2.VideoCapture(0)
    if not cap.isOpened():
        print("Camera not found. Please ensure it's connected.")
        return

    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)

    # For auto-tracking, we might use color detection (e.g., bright orange ball)
    # instead of manual ROI selection since the robot is autonomous.

    # We'll use a simple background subtractor to detect a thrown object
    fgbg = cv2.createBackgroundSubtractorMOG2(history=500, varThreshold=50, detectShadows=False)

    trajectory = []
    tracking = False

    print("Robot ready. Waiting for object...")

    try:
        while True:
            ret, frame = cap.read()
            if not ret:
                break

            # Basic motion detection
            fgmask = fgbg.apply(frame)
            contours, _ = cv2.findContours(fgmask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

            largest_contour = None
            max_area = 0
            for cnt in contours:
                area = cv2.contourArea(cnt)
                if 500 < area < 10000: # Filter noise and huge changes
                    if area > max_area:
                        max_area = area
                        largest_contour = cnt

            if largest_contour is not None:
                (x, y, w, h) = cv2.boundingRect(largest_contour)
                center_x = x + w//2
                center_y = y + h//2

                # Only track if it's high up (being thrown)
                if center_y < 240 or tracking:
                    tracking = True
                    trajectory.append((center_x, center_y))
                    cv2.circle(frame, (center_x, center_y), 5, (0, 255, 0), -1)

                    if len(trajectory) > 20:
                        trajectory.pop(0)

                    predicted_x = predict_landing(trajectory)

                    if predicted_x:
                        # Draw prediction
                        cv2.circle(frame, (predicted_x, 480), 20, (0, 0, 255), -1)

                        # --- Robot Control Logic ---
                        # Frame center is 320. We want the predicted_x to be at 320.
                        error = predicted_x - 320

                        # Simple proportional control
                        if error > 50:
                            # Object landing to the right, move right
                            move_robot(1, 0)
                        elif error < -50:
                            # Object landing to the left, move left
                            move_robot(-1, 0)
                        else:
                            # It's going to land in the center, stay still or move forward slightly
                            # to ensure catching.
                            stop_motors()
            else:
                # Object lost or landed, reset
                if tracking:
                    stop_motors()
                tracking = False
                trajectory.clear()

            # For headless Pi, comment out imshow
            # cv2.imshow('Robot View', frame)

            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

    except KeyboardInterrupt:
        print("Interrupted by user")
    finally:
        stop_motors()
        if GPIO_AVAILABLE:
            GPIO.cleanup()
        cap.release()
        cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
